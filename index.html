from pathlib import Path from IPython.display import Markdown, display s = Path("/mnt/data/Gildea-Home-Merritt-PHONE-FIX.html").read_text(encoding="utf-8") # Split only at existing spaces/newlines so pasting the chunks back-to-back # cannot cut through a word, tag name, CSS token, or JS identifier. # We choose boundaries near equal sizes and INCLUDE the original whitespace # in the preceding chunk, so no content is lost. n_parts = 5 target = len(s) // n_parts cuts = [0] for k in range(1, n_parts): ideal = k * target # Find nearest existing whitespace around the ideal point. candidates = [] for d in range(0, 5000): for pos in (ideal - d, ideal + d): if cuts[-1] < pos < len(s) and s[pos].isspace(): candidates.append(pos + 1) # keep whitespace in preceding chunk break if candidates: break cuts.append(candidates[0]) cuts.append(len(s)) parts = [s[cuts[i]:cuts[i+1]] for i in range(n_parts)] # Render ordinary copyable Markdown code boxes. for i, part in enumerate(parts, 1): display(Markdown(f"### PART {i} OF {n_parts}\n```html\n{part}\n```")) print("All 5 parts above together equal the complete replacement index.html.")
